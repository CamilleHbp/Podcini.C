cmake_minimum_required(VERSION 3.22)
project(podcini_tags LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature APE ASF DSF MATROSKA MOD RIFF SHORTEN TRUEAUDIO)
    set(WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(../../../../third_party/taglib-2.3.2 taglib)
set(TAGLIB_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/taglib-2.3.2/taglib)
include_directories(${TAGLIB_ROOT} ${TAGLIB_ROOT}/toolkit ${TAGLIB_ROOT}/mpeg
    ${TAGLIB_ROOT}/mpeg/id3v2 ${TAGLIB_ROOT}/mpeg/id3v2/frames ${TAGLIB_ROOT}/flac
    ${TAGLIB_ROOT}/ogg ${TAGLIB_ROOT}/ogg/vorbis ${TAGLIB_ROOT}/ogg/opus ${TAGLIB_ROOT}/mp4
    ${CMAKE_CURRENT_BINARY_DIR}/taglib)
if(ANDROID)
    add_library(podcini_tags SHARED metadata_jni.cpp)
    target_link_libraries(podcini_tags PRIVATE tag)
    target_link_options(podcini_tags PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(tag_fixture metadata_fixture.cpp)
    target_link_libraries(tag_fixture PRIVATE tag)
endif()
